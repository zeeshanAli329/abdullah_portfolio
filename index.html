-- ===================================================
-- Music Store Exploratory Data Analysis
-- Author: Muhammad Abdullah
-- Database: PostgreSQL
-- ===================================================

-- 1. Top 10 Best Selling Artists by Total Revenue
WITH ArtistSales AS (
    SELECT 
        art.artist_id,
        art.name AS artist_name,
        SUM(il.unit_price * il.quantity) AS total_spent
    FROM invoice_line il
    JOIN track t ON t.track_id = il.track_id
    JOIN album alb ON alb.album_id = t.album_id
    JOIN artist art ON art.artist_id = alb.artist_id
    GROUP BY art.artist_id, art.name
)
SELECT 
    artist_name,
    ROUND(total_spent::numeric, 2) AS total_revenue,
    DENSE_RANK() OVER (ORDER BY total_spent DESC) AS revenue_rank
FROM ArtistSales
LIMIT 10;

-- 2. Customer Seniority & Total Lifetime Value (LTV)
SELECT 
    c.customer_id,
    c.first_name || ' ' || c.last_name AS customer_name,
    c.country,
    COUNT(DISTINCT i.invoice_id) AS total_orders,
    SUM(i.total) AS total_lifetime_value,
    ROUND(AVG(i.total)::numeric, 2) AS avg_order_value
FROM customer c
JOIN invoice i ON c.customer_id = i.customer_id
GROUP BY c.customer_id, customer_name, c.country
HAVING SUM(i.total) > 100
ORDER BY total_lifetime_value DESC;

-- 3. Most Popular Genre per Country (Using Window Functions)
WITH GenreByCountry AS (
    SELECT 
        c.country,
        g.name AS genre_name,
        COUNT(il.invoice_line_id) AS total_purchases,
        ROW_NUMBER() OVER(PARTITION BY c.country ORDER BY COUNT(il.invoice_line_id) DESC) AS row_num
    FROM invoice_line il
    JOIN invoice i ON i.invoice_id = il.invoice_id
    JOIN customer c ON c.customer_id = i.customer_id
    JOIN track t ON t.track_id = il.track_id
    JOIN genre g ON g.genre_id = t.genre_id
    GROUP BY c.country, g.name
)
SELECT country, genre_name, total_purchases
FROM GenreByCountry
WHERE row_num = 1
ORDER BY total_purchases DESC;
